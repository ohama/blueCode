```fsharp
module BlueCode.Cli.PlanGate

open System
open Spectre.Console
open BlueCode.Core.Domain

/// Outcome of the user's a/r/e/q decision at the plan approval gate.
/// Edit carries the user's comment for inclusion in the next runPlanTurn invocation.
type PlanGateDecision =
    | Accept
    | Reject
    | Edit of comment: string
    | Quit

/// Abstraction over keystroke input so PlanGateTests can supply scripted
/// inputs without invoking the real Console. Production wiring uses
/// realKeyReader below; tests construct a record literal.
type IKeyReader =
    /// Returns the lower-case keystroke character ('a','r','e','q' or other).
    /// May block until a key is pressed.
    abstract member ReadKey : unit -> char
    /// Reads a full line (used after Edit keystroke to capture the comment).
    abstract member ReadLine : unit -> string

/// Production reader: blocking Console.ReadKey + Console.ReadLine.
/// CLAUDE.md note: ReadKey(intercept=true) prevents the typed char from
/// being echoed; we manually echo via stdout for clarity.
let realKeyReader : IKeyReader =
    { new IKeyReader with
        member _.ReadKey () =
            let info = Console.ReadKey(intercept = true)
            Char.ToLowerInvariant info.KeyChar
        member _.ReadLine () =
            // Use Console.In rather than Console.ReadLine() so SetIn redirection in tests works.
            Console.In.ReadLine() |> Option.ofObj |> Option.defaultValue "" }

/// Render a Plan as a Spectre.Console table to stdout. PUBLIC for testability.
/// Columns: # / tool / input (preview, max 60 chars) / rationale.
/// Top-line rationale prints separately above the table.
let render (plan: Plan) : unit =
    // Top rationale (printfn so tests that redirect Console.SetOut can capture it).
    printfn "Proposed plan: %s" plan.Rationale

    let table = Table()
    table.AddColumn("#") |> ignore
    table.AddColumn("tool") |> ignore
    table.AddColumn("input") |> ignore
    table.AddColumn("rationale") |> ignore

    plan.Steps
    |> List.iteri (fun i step ->
        let (ToolName name) = step.Tool
        let (ToolInput m) = step.Input
        let raw = m |> Map.tryFind "_raw" |> Option.defaultValue "{}"
        let preview =
            if raw.Length > 60 then raw.Substring(0, 60) + "..."
            else raw
        table.AddRow([| string (i + 1); name; preview; step.Rationale |]) |> ignore)

    AnsiConsole.Write(table)

    // The prompt itself is plain printfn for stdout-redirect testability.
    printfn ""
    printfn "[a]ccept / [r]eject / [e]dit / [q]uit"

/// Loop until a recognized keystroke is received, then return the decision.
/// Unknown keys re-prompt without exiting. PUBLIC entry point.
let rec promptUser (reader: IKeyReader) : PlanGateDecision =
    let key = reader.ReadKey ()
    printfn "" // newline after intercepted keystroke
    match key with
    | 'a' ->
        printfn "Accepted."
        Accept
    | 'r' ->
        printfn "Rejected — re-prompting LLM."
        Reject
    | 'q' ->
        printfn "Quit."
        Quit
    | 'e' ->
        printfn "Edit — type comment then press Enter:"
        let comment = reader.ReadLine ()
        Edit comment
    | _ ->
        printfn "Unrecognized keystroke. Press a/r/e/q."
        promptUser reader
```

let exitCode =
    if isPlanMode then
        // ─── Plan mode (Phase 16-02) ─────────────────────────────────────
        // Single-turn plan-then-execute. priorSteps comes from session.Steps
        // (resumed or fresh — Program.fs:116-140 already resolved Session).
        let prompt = String.concat " " promptWords
        let model =
            opts.ForcedModel
            |> Option.defaultValue Qwen122B  // 122B canonical default
        let ct = System.Threading.CancellationToken.None

        // Reject loop: track edit-comment between attempts. Bounded by max 3
        // user-facing reject iterations to avoid infinite re-prompting on flaky LLM.
        // (Internal runPlanTurn 2-attempt retry stacks ON TOP of this user-facing loop.)
        let maxUserRejects = 3
        let mutable rejectCount = 0
        let mutable currentPrompt = prompt
        let mutable currentSuffix = planSystemPromptSuffix
        let mutable decision : PlanGate.PlanGateDecision option = None
        let mutable lastError : AgentError option = None

        while decision = None && rejectCount < maxUserRejects do
            let planResult =
                (BlueCode.Core.AgentLoop.runPlanTurn
                    components.Config
                    components.LlmClient
                    model
                    session.Steps
                    currentPrompt
                    currentSuffix
                    ct).GetAwaiter().GetResult()

            match planResult with
            | Error e ->
                eprintfn "%s" (renderError e)
                lastError <- Some e
                decision <- Some PlanGate.Quit  // exit non-zero below
            | Ok plan ->
                PlanGate.render plan
                match PlanGate.promptUser PlanGate.realKeyReader with
                | PlanGate.Accept ->
                    decision <- Some PlanGate.Accept
                | PlanGate.Quit ->
                    decision <- Some PlanGate.Quit
                | PlanGate.Reject ->
                    rejectCount <- rejectCount + 1
                    // [PLAN REJECTED] message embedded in next-turn prompt prefix.
                    // Role=User per Phase 20-03 invariant — the marker carries authority.
                    currentPrompt <- sprintf "[PLAN REJECTED] The previous plan was rejected by the user. Propose a different plan.\n\n%s" prompt
                    // currentSuffix unchanged — still plan mode
                | PlanGate.Edit comment ->
                    rejectCount <- rejectCount + 1
                    currentPrompt <- sprintf "[PLAN EDIT NOTE: %s] Revise the previous plan accordingly.\n\n%s" comment prompt

        match decision, lastError with
        | Some PlanGate.Accept, _ ->
            // Execute the accepted plan by calling runSession with the ORIGINAL prompt.
            // We do NOT pass the validated plan steps directly to the executor — runSession
            // re-asks the LLM to act, and the prompt + priorSteps + system prompt drive it
            // toward the same actions. This matches PLAN-04 acceptance semantics: the user
            // approved the SHAPE; the executor produces tool calls that match.
            let (code, newSteps) =
                (Repl.runSingleTurn prompt session.Steps components renderMode).GetAwaiter().GetResult()
            // Save session — same shape as existing single-turn save at line 156-167.
            let updated =
                { session with
                    Steps = session.Steps @ newSteps
                    LastActivityAt = DateTimeOffset.UtcNow }
            let saveCt = System.Threading.CancellationToken.None
            let saveRes = (components.SessionStore.Save updated saveCt).GetAwaiter().GetResult()
            match saveRes with
            | Ok () -> ()
            | Error e ->
                eprintfn "WARNING: session save failed: %A" e
                Log.Warning("Session save failed: {Error}", sprintf "%A" e)
            code
        | Some PlanGate.Quit, Some _ ->
            // Plan-mode error path (runPlanTurn returned Error).
            1
        | Some PlanGate.Quit, None ->
            // User typed q.
            0
        | _ ->
            // Reject loop exhausted without acceptance.
            eprintfn "Plan-mode: %d rejections without acceptance — aborting." rejectCount
            1
    else
        // ─── Existing dispatch (unchanged from current Program.fs:146-168) ───
        match promptWords with
        | [] ->
            (Repl.runMultiTurnWithSession components renderMode session components.SessionStore).GetAwaiter().GetResult()
        | words ->
            let prompt = String.concat " " words
            let (code, newSteps) =
                (Repl.runSingleTurn prompt session.Steps components renderMode).GetAwaiter().GetResult()

                { session with
                    Steps = session.Steps @ newSteps
                    LastActivityAt = DateTimeOffset.UtcNow }

            | Ok () -> ()
            | Error e ->
                eprintfn "WARNING: session save failed: %A" e
                Log.Warning("Session save failed: {Error}", sprintf "%A" e)

**Why this dispatch shape:**
- `runPlanTurn` (Core, 16-01) is called inside the user-reject loop. Each iteration re-invokes the LLM; the user sees a fresh plan after each reject/edit.
- On accept, we delegate to existing `Repl.runSingleTurn` with the ORIGINAL `prompt` (not the rejected/edited variants). This preserves PERSIST-01 semantics — the executor sees the user's literal request, not the planner's intermediate scaffolding.
- The `[PLAN REJECTED]` and `[PLAN EDIT NOTE: ...]` strings are PREFIXED to the next runPlanTurn prompt — they appear in the user-message turn the LLM reads. Role=User is implicit (single-turn `userInput` is always Role=User in `buildMessages` at AgentLoop.fs:210). Phase 20-03 invariant satisfied.
- `maxUserRejects = 3` is a defensive cap on the user-reject loop. Distinct from the 2-attempt retry inside runPlanTurn (which handles parse/validation failures).
- Plan execution is intentionally NOT step-by-step direct dispatch from `Plan.Steps` — per CLAUDE.md and out-of-scope guardrails, per-step approval is OOS for v2.0; the user approved the SHAPE, then the LLM drives execution.

/// Build a scripted IKeyReader that returns chars from `keys` in order
/// for ReadKey, and lines from `lines` in order for ReadLine.
/// Exhaustion of either list is a programming error in the test (returns
/// space char / empty string and the test should fail by assertion).
let private scriptedReader (keys: char list) (lines: string list) : IKeyReader =
    let mutable kq = keys
    let mutable lq = lines
    { new IKeyReader with
        member _.ReadKey () =
            match kq with
            | [] -> ' '
            | k :: rest -> kq <- rest; k
        member _.ReadLine () =
            match lq with
            | [] -> ""
            | l :: rest -> lq <- rest; l }

let private samplePlan : Plan =
    { Steps = [
        makePlannedStep "read_file" "{\"path\":\"a.fs\"}" "read first"
        makePlannedStep "list_dir" "{\"path\":\"src\"}" "then list"
      ]
      Rationale = "explore then enumerate" }

let private withCapturedStdout (f: unit -> 'a) : 'a * string =
    let prev = Console.Out
    use sw = new StringWriter()
    Console.SetOut(sw)
    try
        let result = f ()
        Console.Out.Flush()
        (result, sw.ToString())
    finally
        Console.SetOut(prev)

let tests =
    testSequenced (testList "PlanGateTests" [

        testCase "promptUser: 'a' -> Accept" <| fun () ->
            let reader = scriptedReader [ 'a' ] []
            let (decision, stdout) = withCapturedStdout (fun () -> promptUser reader)
            Expect.equal decision Accept "decision is Accept"
            Expect.stringContains stdout "Accepted." "stdout reports Accepted"

        testCase "promptUser: 'r' -> Reject" <| fun () ->
            let reader = scriptedReader [ 'r' ] []
            let (decision, stdout) = withCapturedStdout (fun () -> promptUser reader)
            Expect.equal decision Reject "decision is Reject"
            Expect.stringContains stdout "Rejected" "stdout reports Rejected"

        testCase "promptUser: 'q' -> Quit" <| fun () ->
            let reader = scriptedReader [ 'q' ] []

            Expect.equal decision Quit "decision is Quit"

        testCase "promptUser: 'e' captures comment via ReadLine" <| fun () ->
            let reader = scriptedReader [ 'e' ] [ "use grep_search instead of read_file" ]
            let (decision, _) = withCapturedStdout (fun () -> promptUser reader)
            match decision with
            | Edit comment ->

            | other -> failtestf "expected Edit, got %A" other
