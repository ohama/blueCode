**Step 2.** Add new `mt_mode()` function to `bench/run.sh`. Insert it AFTER `b2_mode()` (search for `b2_mode()` in the file) and BEFORE `gate()`. The function runs MT_32b + MT_72b multi-turn fixtures.

EXACT new function to add:

```bash
# ---------------------------------------------------------------------------
# mt_mode() — Multi-turn fixture: turn 1 lists files, turn 2 references turn 1.
# Validates PERSIST-01 end-to-end via the bench harness. Each fixture is two
# blueCode invocations (turn 1 starts a session; turn 2 --resume <id>).
# Adds ~30s on 32B, ~60s on 72B per fixture.
# ---------------------------------------------------------------------------
mt_mode() {
  echo "######## MULTI-TURN: 2 fixtures (MT_32b, MT_72b) ########" | tee -a "$LOG_DIR/timeline.txt"
  for model in 32b 72b; do
    local label="MT_${model}"
    local out="$LOG_DIR/${label}.log"
    local meta="$LOG_DIR/${label}.meta"
    local start_ts=$(date +%s)
    echo "===== $label (model=$model) =====" | tee -a "$LOG_DIR/timeline.txt"

    # Turn 1: capture session id from stderr.
    /usr/bin/time -p dotnet run --project src/BlueCode.Cli --  --verbose --model "$model" --new-session "list 2 files in src/BlueCode.Core" >> "$out" 2> "$out.t1.err"
    local turn1_rc=$?
    cat "$out.t1.err" >> "$out"
    local sid
    sid=$(grep -E "^Session: " "$out.t1.err" | head -1 | awk '{print $2}')
    if [ -z "$sid" ]; then
      echo "label=$label model=$model exit=2 elapsed=0s note=missing-session-id" > "$meta"
      echo "  -> FAIL: turn 1 produced no Session id" | tee -a "$LOG_DIR/timeline.txt"
      continue
    fi
    echo "PROMPT-T1: list 2 files in src/BlueCode.Core" >> "$out"
    echo "SESSION_ID: $sid" >> "$out"

    # Turn 2: --resume the session and ask a follow-up that requires prior context.
    /usr/bin/time -p dotnet run --project src/BlueCode.Cli -- --verbose --model "$model" --resume "$sid" "what was the first filename you listed" >> "$out" 2> "$out.t2.err"
    local turn2_rc=$?
    cat "$out.t2.err" >> "$out"
    echo "PROMPT-T2: what was the first filename you listed" >> "$out"

    local end_ts=$(date +%s)
    local elapsed=$((end_ts - start_ts))
    # Composite exit code: 0 only if BOTH turns exited 0.
    local exit_code=$(( turn1_rc != 0 ? turn1_rc : turn2_rc ))
    echo "label=$label model=$model exit=$exit_code elapsed=${elapsed}s session_id=$sid" > "$meta"
    echo "  -> exit=$exit_code elapsed=${elapsed}s" | tee -a "$LOG_DIR/timeline.txt"
  done
}
```
